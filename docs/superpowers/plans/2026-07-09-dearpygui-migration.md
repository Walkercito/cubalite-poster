```python
"""Loads bundled Geist faces into DearPyGUI's font registry."""

from pathlib import Path

import dearpygui.dearpygui as dpg

from cubalite_poster.config import constants


class FontManager:
    """Registers UI fonts and exposes font item ids for binding."""

    def __init__(self, fonts_dir: Path) -> None:
        self._fonts_dir = fonts_dir
        self._base_font: int | str | None = None
        self._heading_font: int | str | None = None

    @property
    def base_font(self) -> int | str | None:
        return self._base_font

    @property
    def heading_font(self) -> int | str | None:
        return self._heading_font

    def load(self) -> None:
        """Create a font registry and load Geist faces when files exist.

        On missing files, leaves base/heading as None so the app keeps the
        DearPyGUI default font instead of crashing.
        """
        base_path = self._fonts_dir / constants.BASE_FONT_FILE
        heading_path = self._fonts_dir / constants.HEADING_FONT_FILE

        if not base_path.is_file() and not heading_path.is_file():
            return

        with dpg.font_registry():
            if base_path.is_file():
                self._base_font = dpg.add_font(
                    str(base_path),
                    constants.BASE_FONT_SIZE,
                    tag=constants.TAG_BASE_FONT,
                )
            if heading_path.is_file():
                self._heading_font = dpg.add_font(
                    str(heading_path),
                    constants.HEADING_FONT_SIZE,
                    tag=constants.TAG_HEADING_FONT,
                )

        if self._base_font is not None:
            dpg.bind_font(self._base_font)

    def bind_heading(self, item: int | str) -> None:
        """Bind the heading face to a single item when available."""
        if self._heading_font is not None:
            dpg.bind_item_font(item, self._heading_font)
```

```python
"""Fixed dark theme for the application shell."""

import dearpygui.dearpygui as dpg

from cubalite_poster.config import constants


def apply_dark_theme() -> int | str:
    """Build the global dark theme, bind it, and return its id."""
    with dpg.theme(tag=constants.TAG_GLOBAL_THEME) as theme_id:
        with dpg.theme_component(dpg.mvAll):
            dpg.add_theme_color(
                dpg.mvThemeCol_WindowBg,
                constants.COLOR_WINDOW_BG,
                category=dpg.mvThemeCat_Core,
            )
            dpg.add_theme_color(
                dpg.mvThemeCol_ChildBg,
                constants.COLOR_CHILD_BG,
                category=dpg.mvThemeCat_Core,
            )
            dpg.add_theme_color(
                dpg.mvThemeCol_Text,
                constants.COLOR_TEXT,
                category=dpg.mvThemeCat_Core,
            )
            dpg.add_theme_color(
                dpg.mvThemeCol_Button,
                constants.COLOR_BUTTON,
                category=dpg.mvThemeCat_Core,
            )
            dpg.add_theme_color(
                dpg.mvThemeCol_ButtonHovered,
                constants.COLOR_BUTTON_HOVERED,
                category=dpg.mvThemeCat_Core,
            )
            dpg.add_theme_color(
                dpg.mvThemeCol_ButtonActive,
                constants.COLOR_BUTTON_ACTIVE,
                category=dpg.mvThemeCat_Core,
            )
            dpg.add_theme_color(
                dpg.mvThemeCol_Border,
                constants.COLOR_BORDER,
                category=dpg.mvThemeCat_Core,
            )
            dpg.add_theme_color(
                dpg.mvThemeCol_FrameBg,
                constants.COLOR_FRAME_BG,
                category=dpg.mvThemeCat_Core,
            )
            dpg.add_theme_style(
                dpg.mvStyleVar_FrameRounding,
                constants.FRAME_ROUNDING,
                category=dpg.mvThemeCat_Core,
            )
            dpg.add_theme_style(
                dpg.mvStyleVar_WindowBorderSize,
                constants.WINDOW_BORDER_SIZE,
                category=dpg.mvThemeCat_Core,
            )
            dpg.add_theme_style(
                dpg.mvStyleVar_WindowPadding,
                constants.WINDOW_PADDING_X,
                constants.WINDOW_PADDING_Y,
                category=dpg.mvThemeCat_Core,
            )
            dpg.add_theme_style(
                dpg.mvStyleVar_ItemSpacing,
                constants.ITEM_SPACING_X,
                constants.ITEM_SPACING_Y,
                category=dpg.mvThemeCat_Core,
            )

    dpg.bind_theme(theme_id)
    return theme_id


def build_accent_theme() -> int | str:
    """Theme for the primary action button (does not bind globally)."""
    with dpg.theme(tag=constants.TAG_ACCENT_THEME) as theme_id:
        with dpg.theme_component(dpg.mvButton):
            dpg.add_theme_color(
                dpg.mvThemeCol_Button,
                constants.COLOR_ACCENT,
                category=dpg.mvThemeCat_Core,
            )
            dpg.add_theme_color(
                dpg.mvThemeCol_ButtonHovered,
                constants.COLOR_ACCENT_HOVERED,
                category=dpg.mvThemeCat_Core,
            )
            dpg.add_theme_color(
                dpg.mvThemeCol_ButtonActive,
                constants.COLOR_ACCENT_ACTIVE,
                category=dpg.mvThemeCat_Core,
            )
            dpg.add_theme_color(
                dpg.mvThemeCol_Text,
                constants.COLOR_ACCENT_TEXT,
                category=dpg.mvThemeCat_Core,
            )
    return theme_id
```

```python
import dearpygui.dearpygui as dpg

from cubalite_poster.config import constants
from cubalite_poster.ui.fonts import FontManager
from cubalite_poster.ui.theme import build_accent_theme


class MainWindow:
    """Root view: greeting and accent action button.

    A view is responsible only for building and wiring its own widgets. It
    holds no poster-generation logic — that belongs in ``core``.
    """

    def __init__(self, fonts: FontManager) -> None:
        self._fonts = fonts
        self._build()

    def _build(self) -> None:
        with dpg.window(
            tag=constants.TAG_PRIMARY_WINDOW,
            no_title_bar=True,
            no_resize=True,
            no_move=True,
            no_collapse=True,
            no_close=True,
        ):
            dpg.add_spacer(height=constants.CONTENT_TOP_SPACER)
            with dpg.group(horizontal=True):
                dpg.add_spacer(width=constants.CONTENT_LEFT_SPACER)
                with dpg.group():
                    dpg.add_text(
                        constants.GREETING_TEXT,
                        tag=constants.TAG_GREETING,
                    )
                    self._fonts.bind_heading(constants.TAG_GREETING)

                    dpg.add_spacer(height=constants.WIDGET_SPACING)

                    dpg.add_button(
                        label=constants.TOUCH_ME_BUTTON_TEXT,
                        tag=constants.TAG_TOUCH_ME_BUTTON,
                        callback=self._on_touch_me,
                    )
                    dpg.bind_item_theme(
                        constants.TAG_TOUCH_ME_BUTTON,
                        build_accent_theme(),
                    )

    @staticmethod
    def _on_touch_me() -> None:
```

```python
"""Application shell: owns the DearPyGUI context, viewport and lifecycle."""

import dearpygui.dearpygui as dpg

from cubalite_poster import resources
from cubalite_poster.config import constants
from cubalite_poster.ui.fonts import FontManager
from cubalite_poster.ui.main_window import MainWindow
from cubalite_poster.ui.theme import apply_dark_theme


class App:
    """Wires fonts, theme, main view and the viewport run loop."""

    def __init__(self) -> None:
        dpg.create_context()

        self._fonts = FontManager(resources.FONTS_DIR)
        self._fonts.load()
        apply_dark_theme()
        MainWindow(self._fonts)

        self._configure_viewport()

    def _configure_viewport(self) -> None:
        dpg.create_viewport(
            title=constants.WINDOW_TITLE,
            width=constants.WINDOW_WIDTH,
            height=constants.WINDOW_HEIGHT,
            resizable=constants.WINDOW_RESIZABLE,
        )
        dpg.setup_dearpygui()
        dpg.show_viewport()
        dpg.set_primary_window(constants.TAG_PRIMARY_WINDOW, True)

    def run(self) -> None:
        dpg.start_dearpygui()
        dpg.destroy_context()


def main() -> None:
    App().run()
```
